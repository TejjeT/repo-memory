from gateway import GatewayClient, TransientError

# Gateway/client retry layer.
GATEWAY_MAX_RETRIES = 0

# Worker-level retry budget for transient settlement submission failures.
SETTLE_MAX_RETRIES = 3


def make_client(gateway):
    return GatewayClient(gateway, max_retries=GATEWAY_MAX_RETRIES)


def settle_batch(client, batch_id, payload, idempotency_key):
    """Submit one settlement batch.

    Transient gateway failures are retried up to SETTLE_MAX_RETRIES times.
    The same idempotency_key is reused on every attempt, so an ambiguous
    failure (settlement committed, response lost) deduplicates instead of
    settling twice. Permanent failures propagate immediately. The
    client-side retry loop stays disabled (GATEWAY_MAX_RETRIES = 0) so
    retries happen in exactly one layer.
    """
    attempts = 0
    while True:
        try:
            return client.submit(batch_id, idempotency_key)
        except TransientError:
            if attempts >= SETTLE_MAX_RETRIES:
                raise
            attempts += 1
